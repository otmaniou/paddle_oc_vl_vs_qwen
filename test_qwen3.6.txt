import gc

DOSSIERS_PAR_BATCH = 4    # 4 dossiers × ~4 pages = batch de 16 pages

deja_traites = {f.stem for f in JSON_DIR.glob('*.json')}
a_traiter    = [p for p in pdfs if p.stem not in deja_traites]
log(f'À traiter : {len(a_traiter)} | déjà traités : {len(deja_traites)}')

t_total = time.time()
n_ok = n_err = 0
TYPES_ATTENDUS = {'OV', 'ANNEXE_I', 'ANNEXE_II', 'BULLETIN'}

def finalize_dossier(pdf_path, pages_raw, t0):
    global n_ok, n_err
    dt = round(time.time() - t0, 2)
    try:
        results, doublons = {}, []
        tok_in = tok_out = 0
        for pr in sorted(pages_raw, key=lambda x: x['index']):
            data = parse_json(pr['text'])
            dt_  = data.get('type', 'INCONNU')
            tok_in  += pr['tokens_in']
            tok_out += pr['tokens_out']
            if dt_ in ('AUTRE', 'INCONNU'): continue
            data = normalise_doc(dt_, data)
            if dt_ in results: doublons.append(dt_); continue
            results[dt_] = data
        pm = sorted(TYPES_ATTENDUS - set(results.keys()))
        result = {
            'fichier':          pdf_path.name,
            'date_traitement':  datetime.now().strftime('%Y-%m-%d %H:%M:%S'),
            'temps_total_s':    dt,
            'tokens_in':        tok_in,
            'tokens_out':       tok_out,
            'tokens_total':     tok_in + tok_out,
            'pages_trouvees':   ', '.join(sorted(results.keys())),
            'pages_manquantes': ', '.join(pm) if pm else None,
            'anomalies':        'DOUBLON: ' + ', '.join(doublons) if doublons else None,
            **{t: results.get(t, {}) for t in TYPES_ATTENDUS},
        }
        with open(JSON_DIR / f'{pdf_path.stem}.json', 'w', encoding='utf-8') as f:
            json.dump(result, f, ensure_ascii=False, indent=2, default=str)
        n_ok += 1
        restants = len(a_traiter) - n_ok - n_err
        eta = (time.time() - t_total) / max(1, n_ok + n_err) * restants
        msg = (f'[{n_ok+n_err:>4}/{len(a_traiter)}] ✅ {pdf_path.name} | {dt:.1f}s'
               f' | tok={tok_in+tok_out} | {result["pages_trouvees"]}')
        if result['pages_manquantes']: msg += f' | ⚠️  {result["pages_manquantes"]}'
        if result['anomalies']:        msg += f' | 🔴 {result["anomalies"]}'
        msg += f' | ETA {eta/3600:.1f}h'
        log(msg)
    except Exception as e:
        n_err += 1
        log(f'[{n_ok+n_err:>4}/{len(a_traiter)}] ❌ {pdf_path.name} — {e}')

# ── Boucle : groupes de 4 dossiers → 1 batch de ~16 pages ──
for i in range(0, len(a_traiter), DOSSIERS_PAR_BATCH):
    group = a_traiter[i:i+DOSSIERS_PAR_BATCH]
    batch_pages, meta = [], {}

    for pdf in group:
        t0 = time.time()
        try:
            pages   = pdf_to_pages(pdf)
            actives = [p for p in pages if not is_blank(p['image'])]
        except Exception as e:
            log(f'❌ Chargement {pdf.name} : {e}')
            n_err += 1
            continue
        meta[pdf] = (t0, [])
        batch_pages.extend((pdf, p) for p in actives)

    if not batch_pages:
        continue

    # Inférence 16 pages, repli automatique 8 si OOM
    try:
        reps = ask_batch(PROMPT_UNIVERSEL, [pg['image'] for _, pg in batch_pages])
    except torch.cuda.OutOfMemoryError:
        torch.cuda.empty_cache()
        log('⚠️  OOM à 16 pages → repli batchs de 8')
        reps = []
        for bs in range(0, len(batch_pages), 8):
            sub = batch_pages[bs:bs+8]
            reps += ask_batch(PROMPT_UNIVERSEL, [pg['image'] for _, pg in sub])

    for (pdf, pg), rep in zip(batch_pages, reps):
        meta[pdf][1].append({
            'index': pg['index'], 'text': rep['text'],
            'tokens_in': rep['tokens_in'], 'tokens_out': rep['tokens_out'],
        })
    gc.collect(); torch.cuda.empty_cache()

    for pdf, (t0, raw) in meta.items():
        finalize_dossier(pdf, raw, t0)

    if i == 0:
        log(f'   VRAM libre après 1er groupe : {torch.cuda.mem_get_info()[0]/1e9:.1f} GB')

# ── Excel + stats ──
log('Génération Excel...')
rows = []
for json_file in sorted(JSON_DIR.glob('*.json')):
    with open(json_file, encoding='utf-8') as f:
        rows.append(json.load(f))
create_excel(EXCEL_PATH, rows)

elapsed = time.time() - t_total
log(f'✅ Terminé en {elapsed:.1f}s ({elapsed/max(1,n_ok):.1f}s/dossier)')
log(f'   Traités : {n_ok} | Erreurs : {n_err}')
