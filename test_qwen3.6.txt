import sys, importlib
from pathlib import Path
import transformers.integrations.finegrained_fp8 as tf_fg

FP8_DIRS = [
    Path('/domino/edv/modelhub/ModelHub-model-huggingface-kernels-community/finegrained-fp8/v4'),    # priorité
    Path('/domino/edv/modelhub/ModelHub-model-huggingface-kernels-community/finegrained-fp8/main'), # fallback
]

def load_kernel_from_dir(path: Path):
    if str(path) not in sys.path:
        sys.path.insert(0, str(path))
    # cas 1 : paquet finegrained_fp8/ dans le dépôt
    try:
        return importlib.import_module('finegrained_fp8')
    except Exception:
        pass
    # cas 2 : la racine du dépôt est elle-même un paquet
    if (path / '__init__.py').exists():
        parent = str(path.parent)
        if parent not in sys.path:
            sys.path.insert(0, parent)
        return importlib.import_module(path.name)
    # cas 3 : un sous-répertoire paquet
    for sub in sorted(path.iterdir()):
        if sub.is_dir() and (sub / '__init__.py').exists():
            return importlib.import_module(sub.name)
    raise ImportError(f'Aucun paquet importable dans {path}')

kernel_mod = None
for path in FP8_DIRS:
    if not path.exists():
        print(f'⚠️  {path.name} : répertoire introuvable'); continue
    try:
        kernel_mod = load_kernel_from_dir(path)
        print(f'✅ Kernel local chargé depuis {path.name}')
        break
    except Exception as e:
        print(f'⚠️  {path.name} : {e}')

if kernel_mod is None:
    raise RuntimeError('❌ Kernel local non chargeable → voir fallback dequantize (cellule 4.6)')

# Injection dans transformers (les noms exacts vus dans ton traceback)
tf_fg._FINEGRAINED_FP8 = kernel_mod
tf_fg._load_finegrained_fp8_kernel = lambda: None

assert hasattr(kernel_mod, 'matmul'), '❌ module sans .matmul — mauvais paquet'
print('✅ Kernel injecté dans transformers — .matmul OK')
