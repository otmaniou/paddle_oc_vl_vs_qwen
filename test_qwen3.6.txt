import sys, types, importlib
from types import SimpleNamespace
from pathlib import Path
import transformers.integrations.finegrained_fp8 as tf_fg

FP8_V4 = Path('/domino/edv/modelhub/ModelHub-model-huggingface-kernels-community/finegrained-fp8/v4')

init_files = sorted(FP8_V4.rglob('__init__.py'))
pkg_parent = init_files[0].parent.parent
if str(pkg_parent) not in sys.path:
    sys.path.insert(0, str(pkg_parent))
kernel_mod = importlib.import_module(init_files[0].parent.name)   # ex 'torch-cuda'

print('package dir :', [n for n in dir(kernel_mod) if not n.startswith('_')])
print('sous-modules:', [n for n in dir(kernel_mod)
                         if isinstance(getattr(kernel_mod, n), types.ModuleType)])

PREFER = ('matmul', 'fp8_matmul', 'finegrained_fp8_matmul',
          'fp8_block_scaled_matmul', 'forward', 'fwd', 'apply')

def find_callable(mod):
    # 1) attribut direct callable (pas un module)
    for name in PREFER:
        f = getattr(mod, name, None)
        if callable(f) and not isinstance(f, types.ModuleType):
            return f, f'.{name}'
    # 2) fonction à l'intérieur des sous-modules (.so)
    for sub_name in dir(mod):
        sub = getattr(mod, sub_name)
        if isinstance(sub, types.ModuleType):
            for name in PREFER:
                f = getattr(sub, name, None)
                if callable(f):
                    return f, f'.{sub_name}.{name}'
            for name in dir(sub):                      # fallback : 1er callable public
                if name.startswith('__'): continue
                f = getattr(sub, name)
                if callable(f):
                    return f, f'.{sub_name}.{name}'
    # 3) n'importe quel callable public au niveau paquet
    for name in dir(mod):
        if name.startswith('_'): continue
        f = getattr(mod, name)
        if callable(f) and not isinstance(f, types.ModuleType):
            return f, f'.{name}'
    return None, None

fn, where = find_callable(kernel_mod)
if fn is None:
    sub = getattr(kernel_mod, 'matmul', None)
    if isinstance(sub, types.ModuleType):
        print('matmul .so dir :', [n for n in dir(sub) if not n.startswith('__')])
    raise RuntimeError('❌ aucune fonction matmul trouvée → envoie les dir() affichés')

print(f'✅ fonction matmul trouvée en {where}')

# Shim exposant .matmul comme l'attend transformers
tf_fg._FINEGRAINED_FP8 = SimpleNamespace(matmul=fn)
tf_fg._load_finegrained_fp8_kernel = lambda: None
print('✅ shim injecté dans transformers')
