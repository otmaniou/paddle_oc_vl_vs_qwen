# 4.5 (v2) — Kernel finegrained-fp8 depuis ModelHub (paquet dans build/)
# ─────────────────────────────────────────────────────────────
import sys, importlib, zipfile
from pathlib import Path
import transformers.integrations.finegrained_fp8 as tf_fg

FP8_DIRS = [
    Path('/domino/edv/modelhub/ModelHub-model-huggingface-kernels-community/finegrained-fp8/v4'),
    Path('/domino/edv/modelhub/ModelHub-model-huggingface-kernels-community/finegrained-fp8/main'),
]

EXTRACT_ROOT = OUTPUT_DIR / 'fp8_kernel_extracted'
EXTRACT_ROOT.mkdir(parents=True, exist_ok=True)
pyver = f'cp{sys.version_info.major}{sys.version_info.minor}'   # cp311 chez toi

def try_import_from_dir(dirpath: Path):
    """Ajoute dirpath au sys.path et importe le 1er paquet Python trouvé."""
    if str(dirpath) not in sys.path:
        sys.path.insert(0, str(dirpath))
    for sub in sorted(dirpath.iterdir()):
        if sub.is_dir() and (sub / '__init__.py').exists():
            return importlib.import_module(sub.name)
    raise ImportError(f'aucun paquet dans {dirpath}')

kernel_mod, used = None, None

for tag, base in [('v4', FP8_DIRS[0]), ('v1', FP8_DIRS[1])]:
    if not base.exists():
        print(f'⚠️  {tag} : répertoire introuvable'); continue

    # ── a) paquet déjà déployé quelque part sous build/ ──
    init_files = sorted(base.rglob('__init__.py'))
    if init_files:
        try:
            kernel_mod = try_import_from_dir(init_files[0].parent.parent)
            used = f'{tag} paquet déployé ({init_files[0].parent.name})'
            break
        except Exception as e:
            print(f'⚠️  {tag} paquet : {e}')

    # ── b) wheels dans build/ → décompression locale (offline) ──
    wheels = sorted(base.rglob('*.whl'), key=lambda w: (pyver not in w.name, w.name))
    for w in wheels:
        try:
            dest = EXTRACT_ROOT / f'{tag}_{w.stem}'
            if not dest.exists():
                with zipfile.ZipFile(w) as z:
                    z.extractall(dest)
            kernel_mod = try_import_from_dir(dest)
            used = f'{tag} wheel {w.name}'
            break
        except Exception as e:
            print(f'⚠️  {tag} wheel {w.name} : {e}')
    if kernel_mod:
        break

if kernel_mod is None:
    for tag, base in [('v4', FP8_DIRS[0]), ('v1', FP8_DIRS[1])]:
        print(f'--- contenu {tag} ---')
        for p in sorted(base.rglob('*')):
            if p.is_file(): print('  ', p.relative_to(base))
    raise RuntimeError('❌ Toujours non chargeable → ENVOIE la liste affichée ci-dessus')

# Injection dans transformers
tf_fg._FINEGRAINED_FP8 = kernel_mod
tf_fg._load_finegrained_fp8_kernel = lambda: None
assert hasattr(kernel_mod, 'matmul'), '❌ module sans .matmul'
print(f'✅ Kernel injecté via {used} — .matmul OK')
